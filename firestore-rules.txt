rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Only you (the admin email below) can change things.
    function isAdmin() {
      return request.auth != null
        && request.auth.token.email == 'YOUR-EMAIL@example.com';
    }

    // Anyone can see products and store settings. Only you can edit them.
    match /products/{id} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /settings/{id} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // Customers can place an order, but only you can read or change orders.
    match /orders/{id} {
      allow create: if request.resource.data.keys().hasAll(['items', 'total', 'customer'])
        && request.resource.data.items.size() > 0
        && request.resource.data.items.size() <= 50;
      allow read, update, delete: if isAdmin();
    }
  }
}
